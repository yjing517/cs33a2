```bash
set -o pipefail
run_dir="reports/ddp_nccl/$(date -u +%Y%m%dT%H%M%S)"
mkdir -p "$run_dir"
for dtype in float32 float64; do
  for repeat in 1 2 3 4 5; do
    stem="$run_dir/$dtype-$repeat"
    command=(.venv/bin/python -m torch.distributed.run --standalone --nnodes=1 --nproc-per-node=2 --module scripts.check_ddp_nccl --dtype "$dtype" --steps 5)
    printf '%q ' env CUDA_VISIBLE_DEVICES=0,1 OMP_NUM_THREADS=1 NCCL_DEBUG=INFO "${command[@]}" > "$stem.command.txt"
    env CUDA_VISIBLE_DEVICES=0,1 OMP_NUM_THREADS=1 NCCL_DEBUG=INFO "${command[@]}" > "$stem.stdout.log" 2> "$stem.stderr.log"
    code=$?
    printf '%s\n' "$code" > "$stem.exit_code.txt"
    if [ "$code" -ne 0 ]; then exit "$code"; fi
  done
done
```
